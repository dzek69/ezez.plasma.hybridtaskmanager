cmake_minimum_required(VERSION 3.22)
project(HybridTaskManager VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ECM 6.20 REQUIRED NO_MODULE)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs6)
include(KDECMakeSettings)
include(KDECompilerSettings NO_POLICY_SCOPE)
include(ECMQtDeclareLoggingCategory)
find_package(Qt6 6.10 REQUIRED COMPONENTS Core Gui Qml Quick Widgets Test)
find_package(KF6 REQUIRED COMPONENTS Config CoreAddons I18n KIO Notifications Service WindowSystem)
find_package(Plasma REQUIRED)
find_package(PlasmaQuick REQUIRED)
find_package(PlasmaActivities REQUIRED)
find_package(PlasmaActivitiesStats REQUIRED)
find_package(KSysGuard REQUIRED)
find_package(LibNotificationManager REQUIRED)
add_subdirectory(src)
find_package(Gettext REQUIRED)
add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/pl.mo
    COMMAND ${GETTEXT_MSGFMT_EXECUTABLE} --check -o ${CMAKE_CURRENT_BINARY_DIR}/pl.mo ${CMAKE_CURRENT_SOURCE_DIR}/po/pl.po
    DEPENDS po/pl.po)
add_custom_target(translations ALL DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/pl.mo)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/pl.mo
    DESTINATION ${KDE_INSTALL_LOCALEDIR}/pl/LC_MESSAGES
    RENAME plasma_applet_org.ezez.plasma.hybridtaskmanager.mo)
include(CTest)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
